

# 后端目录结构（清晰的API分层）


```
backend/
├── src/
│   ├── index.js              # Worker入口文件
│   ├── config/              # 配置
│   │   ├── constants.js     # 常量定义
│   │   ├── env.js          # 环境配置
│   │   └── routes.js       # 路由定义（所有API路由在此注册）
│   ├── api/                # API接口（与前端api/目录一一对应）
│   │   ├── auth/           # 认证接口
│   │   │   ├── login.js
│   │   │   ├── register.js
│   │   │   └── recovery.js
│   │   ├── notes/          # 笔记接口
│   │   │   ├── create.js
│   │   │   ├── read.js
│   │   │   ├── update.js
│   │   │   ├── delete.js
│   │   │   └── list.js
│   │   ├── categories/     # 分类接口
│   │   │   ├── index.js
│   │   │   └── manage.js
│   │   └── sync/           # 同步接口
│   │       ├── push.js     # 推送离线更改
│   │       ├── pull.js     # 拉取服务器数据
│   │       └── conflicts.js # 冲突解决
│   ├── middleware/         # 中间件
│   │   ├── auth.js
│   │   ├── validation.js
│   │   ├── rateLimit.js
│   │   └── cors.js
│   ├── services/          # 业务逻辑服务
│   │   ├── AuthService.js
│   │   ├── NoteService.js
│   │   ├── SyncService.js
│   │   └── BackupService.js
│   ├── storage/           # 数据存储层
│   │   ├── Database.js    # D1封装
│   │   ├── KVStore.js     # KV封装
│   │   └── models/        # 数据模型
│   │       ├── User.js
│   │       ├── Note.js
│   │       └── Category.js
│   └── utils/             # 工具函数
│       ├── encryption.js
│       ├── logger.js
│       └── response.js
├── wrangler.toml
└── package.json

```
